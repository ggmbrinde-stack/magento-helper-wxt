// Paste this into the DevTools console on the attribute-set custom-options edit page.
// Snippet 1: dump likely form/table/fieldset anchors around the edit UI.
(function () {
  'use strict';
  const results = [];
  const anchors = [
    ...document.querySelectorAll('form'),
    ...document.querySelectorAll('fieldset'),
    ...document.querySelectorAll('table.admin__table'),
    ...document.querySelectorAll('div[data-bind*="foreach"]'),
    ...document.querySelectorAll('div[data-bind*="template"]'),
    ...document.querySelectorAll('div[class*="areas"]'),
    ...document.querySelectorAll('div[class*="options"]'),
  ];
  for (const node of anchors) {
    if (!node || !document.body.contains(node)) continue;
    results.push({
      tag: node.tagName.toLowerCase(),
      id: node.id || '',
      classes: (node.className && typeof node.className === 'string' ? node.className : '')
        .split(/\s+/)
        .filter(Boolean)
        .slice(0, 8)
        .join(' '),
      dataBind: node.getAttribute ? (node.getAttribute('data-bind') || '') : '',
      children: Math.max(0, node.children.length),
      snippet: (node.textContent || '')
        .replace(/\s+/g, ' ')
        .slice(0, 160),
    });
  }
  results.sort((a, b) => (b.children || 0) - (a.children || 0));
  console.group('[mhx opt probe 1] form/table/fieldset anchors');
  console.log('anchor count:', results.length);
  console.table(
    results.slice(0, 24).map((r) => ({
      tag: r.tag,
      id: r.id || '—',
      classes: r.classes || '—',
      dataBind: r.dataBind || '',
      children: String(r.children),
      snippet: r.snippet || '',
    })),
  );
  console.groupEnd();
})();
